<script>
(function(){
  var cur=0,total=3,timer;
  function goSlide(n){
    document.querySelectorAll('.hs-slide').forEach(function(s,i){s.classList.toggle('active',i===n);});
    document.querySelectorAll('.hs-dot').forEach(function(d,i){d.classList.toggle('active',i===n);});
    cur=n; resetProgress();
  }
  function changeSlide(dir){goSlide((cur+dir+total)%total);}
  function resetProgress(){
    var f=document.getElementById('hs-fill');
    if(!f)return;
    f.style.transition='none'; f.style.width='0%';
    requestAnimationFrame(function(){requestAnimationFrame(function(){
      f.style.transition='width 5s linear'; f.style.width='100%';
    });});
    clearInterval(timer);
    timer=setInterval(function(){changeSlide(1);},5000);
  }
  var startX=0,slider=document.getElementById('hero-slider');
  if(slider){
    slider.addEventListener('touchstart',function(e){startX=e.touches[0].clientX;},{passive:true});
    slider.addEventListener('touchend',function(e){var d=startX-e.changedTouches[0].clientX;if(Math.abs(d)>50)changeSlide(d>0?1:-1);},{passive:true});
    slider.addEventListener('mouseenter',function(){clearInterval(timer);});
    slider.addEventListener('mouseleave',function(){timer=setInterval(function(){changeSlide(1);},5000);});
  }
  window.AN_goSlide=goSlide; window.AN_changeSlide=changeSlide;
  resetProgress();
})();
</script>
